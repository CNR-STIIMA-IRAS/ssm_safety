cmake_minimum_required(VERSION 3.16.0)
project(velocity_scaling_iso15066)


find_package(rdyn_core REQUIRED)


option(ENABLE_TESTING "ENABLE TESTING" ON)


if (ENABLE_TESTING)
  find_package(GTest REQUIRED)
  include(GoogleTest)
  include(CTest)
endif()


add_library(velocity_scaling SHARED
  src/${PROJECT_NAME}/ssm15066.cpp
  )

target_link_libraries(velocity_scaling PUBLIC
  rdyn_core::rdyn_core)

set_target_properties(velocity_scaling PROPERTIES
            CXX_STANDARD 17
            CXX_EXTENSIONS OFF
            )
target_compile_options(velocity_scaling PRIVATE
  -funroll-loops -Ofast
  )
target_include_directories(velocity_scaling
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/include> #if BUILD_INTERFACE, add ${CMAKE_SOURCE_DIR}/include (headers taken by source directory)
    $<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/include> #if BUILD_INTERFACE, add ${CMAKE_SOURCE_DIR}/include (headers taken by autogenerated commands)
    $<INSTALL_INTERFACE:include> # if INSTALL_INTERFACE, add header from install path
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
  )
add_library(velocity_scaling::velocity_scaling ALIAS velocity_scaling)
