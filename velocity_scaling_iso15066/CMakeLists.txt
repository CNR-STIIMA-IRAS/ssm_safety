cmake_minimum_required(VERSION 3.16.0)
project(velocity_scaling_iso15066)

find_package(pinocchio REQUIRED)

option(ENABLE_TESTING "ENABLE TESTING" ON)

if (ENABLE_TESTING)
  find_package(GTest REQUIRED)
  include(GoogleTest)
  include(CTest)
endif()

add_library(velocity_scaling_iso15066 SHARED
  src/${PROJECT_NAME}/ssm15066.cpp
  src/${PROJECT_NAME}/ssm_fixed_areas.cpp
  src/${PROJECT_NAME}/ssm_fixed_distance.cpp
  src/${PROJECT_NAME}/ssm_base.cpp
  )

target_link_libraries(velocity_scaling_iso15066 PUBLIC
  pinocchio::pinocchio)

set_target_properties(velocity_scaling_iso15066 PROPERTIES
            CXX_STANDARD 17
            CXX_EXTENSIONS OFF
            )
target_compile_options(velocity_scaling_iso15066 PRIVATE
  -funroll-loops -Ofast 
  )

target_include_directories(velocity_scaling_iso15066
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/include> #if BUILD_INTERFACE, add ${CMAKE_SOURCE_DIR}/include (headers taken by source directory)
    $<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/include> #if BUILD_INTERFACE, add ${CMAKE_SOURCE_DIR}/include (headers taken by autogenerated commands)
    $<INSTALL_INTERFACE:include> # if INSTALL_INTERFACE, add header from install path
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
  )
add_library(velocity_scaling_iso15066::velocity_scaling_iso15066 ALIAS velocity_scaling_iso15066)

# #############################################################################
# Testing  ##
# #############################################################################
if(ENABLE_TESTING)

  set(SSM_TEST_SOURCES
    test/test_initialization.cpp
    test/test_scaling_bounds.cpp
    test/test_performance_large_cloud.cpp
    test/test_ssm_iso150666.cpp
  )

  foreach(test_src ${SSM_TEST_SOURCES})
    get_filename_component(test_name ${test_src} NAME_WE)  # get base name without extension

    add_executable(${test_name} ${test_src})
    target_link_libraries(${test_name}
      PUBLIC
      velocity_scaling_iso15066::velocity_scaling_iso15066
      GTest::gtest_main
    )
    target_compile_definitions(${test_name} PRIVATE TEST_DIR="${CMAKE_CURRENT_LIST_DIR}/test")
    target_compile_options(${test_name} PUBLIC $<$<CONFIG:Release>:-Ofast >)

    gtest_discover_tests(${test_name})
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()


endif()



install(DIRECTORY include/${PROJECT_NAME}
  DESTINATION include)
  
install(
  TARGETS velocity_scaling_iso15066 #node_connection_test kdtree_test
  EXPORT velocity_scaling_iso15066Targets
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin
)

install(EXPORT velocity_scaling_iso15066Targets
  DESTINATION "share/${PROJECT_NAME}/cmake"
  NAMESPACE velocity_scaling_iso15066::
  FILE velocity_scaling_iso15066Targets.cmake
)

include(CMakePackageConfigHelpers)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/velocity_scaling_iso15066ConfigVersion.cmake
  VERSION 0.0.1
  COMPATIBILITY AnyNewerVersion)

install(FILES
  ${CMAKE_SOURCE_DIR}/cmake_config/velocity_scaling_iso15066Config.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/velocity_scaling_iso15066ConfigVersion.cmake
  DESTINATION "share/${PROJECT_NAME}/cmake")

# make uninstall
add_custom_target("uninstall" COMMENT "Uninstall installed files")
add_custom_command(
    TARGET "uninstall"
    POST_BUILD
    COMMENT "Uninstall files with install_manifest.txt"
    COMMAND xargs rm -vf < install_manifest.txt || echo Nothing in
            install_manifest.txt to be uninstalled!
)
